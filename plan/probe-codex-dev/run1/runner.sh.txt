#!/usr/bin/env bash
# Live probe of the Codex developer role (glab 0.7.0, 74c5830), run by the
# manager session at the user's instruction of 2026-10-18: Codex's default
# model, effort high, --role developer, on a throwaway git repo (never the
# user's repo); one start, one send, then stop.
set -u
export TMPDIR=<tmp>
S=<home>/projects/skills/plugins/glab/scripts/external-agent.sh
CLI="$HOME/.local/bin/codex"
base=<tmp>/codex-dev-l1; d=$base/out; R=$base/repo
rm -rf "$d" "$R"; mkdir -p "$d" "$R"
marked() { local p e; for p in /proc/[0-9]*; do
  e=$(tr '\0' '\n' <"$p/environ" 2>/dev/null) || continue
  printf '%s\n' "$e" | command grep -qx "GLAB_EXTERNAL_AGENT=$1" && echo "${p#/proc/}"; done; }
# the throwaway repository
( cd "$R" && git init -q -b main && git config user.email t@example.invalid && git config user.name t \
  && printf '# Working agreement\n\nA toy repo for a test. Keep changes small. Do not commit.\n' >AGENTS.md \
  && printf 'def add(a, b):\n    return a + b\n' >calc.py \
  && printf 'calc\n' >README.md && git add -A && git commit -qm init )
git -C "$R" rev-parse HEAD >"$d/head.before"; ( cd "$R" && sha256sum .git/config; ls .git/hooks | sort ) >"$d/git.before"
"$S" check --backend codex --cli "$CLI" >"$d/check.out" 2>"$d/check.err"; echo "check rc=$?" | tee "$d/rc.txt"
cat >"$d/brief.md" <<'B'
Task (a test of the external developer role, in a throwaway repository):
1. In calc.py, add a function `sub(a, b)` that returns `a - b`, below `add`.
2. Create a new file `notes.txt` containing exactly one line: `sub added`.
3. Run `printenv GLAB_EXTERNAL_AGENT` and put its output in your final message, on a line starting `marker:`.
Do not commit, stage or touch .git. Nothing else.
B
SECONDS=0
"$S" start --backend codex --cli "$CLI" --effort high --role developer --label dev-l1 --repo "$R" --timeout 900 --brief "$d/brief.md" >"$d/start.out" 2>"$d/start.err"
echo "start rc=$? ${SECONDS}s" | tee -a "$d/rc.txt"
id=$(command sed -n 's/^agent: \([^ ]*\) .*/\1/p' "$d/start.out" | head -n 1); echo "id=$id" | tee -a "$d/rc.txt"
[ -n "$id" ] || exit 0
echo "marked after start: $(marked "$id" | wc -l)" | tee -a "$d/rc.txt"
( cd "$R" && git status --porcelain=v1 --untracked-files=all ) >"$d/status.after-start"
printf '%s\n' 'Now also add a one-line docstring to `sub`: """Return a minus b.""". Nothing else.' >"$d/msg.md"
SECONDS=0
"$S" send "$id" --message "$d/msg.md" --timeout 900 >"$d/send.out" 2>"$d/send.err"
echo "send rc=$? ${SECONDS}s" | tee -a "$d/rc.txt"
echo "marked after send: $(marked "$id" | wc -l)" | tee -a "$d/rc.txt"
( cd "$R" && git status --porcelain=v1 --untracked-files=all; echo ---; git diff; echo ---; cat notes.txt ) >"$d/status.after-send" 2>&1
git -C "$R" rev-parse HEAD >"$d/head.after"; ( cd "$R" && sha256sum .git/config; ls .git/hooks | sort ) >"$d/git.after"
cmp -s "$d/head.before" "$d/head.after" && echo "HEAD unchanged" | tee -a "$d/rc.txt" || echo "HEAD CHANGED" | tee -a "$d/rc.txt"
cmp -s "$d/git.before" "$d/git.after" && echo "config+hooks unchanged" | tee -a "$d/rc.txt" || echo "config/hooks CHANGED" | tee -a "$d/rc.txt"
"$S" list >"$d/list.out" 2>&1
cp "$TMPDIR/glab-external-agents/$id/"transcript.* "$d/" 2>/dev/null
"$S" stop "$id" >"$d/stop.out" 2>"$d/stop.err"; echo "stop rc=$?" | tee -a "$d/rc.txt"
echo "marked after stop: $(marked "$id" | wc -l)" | tee -a "$d/rc.txt"
echo "agent dir left: $([ -e "$TMPDIR/glab-external-agents/$id" ] && echo yes || echo no)" | tee -a "$d/rc.txt"
git -C <home>/projects/skills status --short -uall >"$d/skills-status.out"
